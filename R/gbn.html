# Modelo no paramétrico
K <- function(x) exp(-(x^2)/2) / sqrt(2*pi)
nw_weight <- function(x0, xi, bw) prod(K((x0 - xi)/bw))

nw_fit <- function(X, y, bw) {
  X <- as.matrix(X); n <- nrow(X)
  ajustados <- numeric(n); hii <- numeric(n)
  for (i in 1:n) {
    w <- apply(X, 1, function(fila) nw_weight(X[i,], fila, bw))
    ajustados[i] <- sum(w*y)/sum(w)
    hii[i] <- w[i]/sum(w)
  }
  list(fitted = ajustados, hii = hii)
}

# Silverman generalizado: el exponente depende de la dimensión del predictor
silverman_bw <- function(x, d = 1) 1.06 * sd(x) * length(x)^(-1/(d+4))
# Modelo no paramétrico
loglik_gauss <- function(rss, n) -(n/2) * (log(2*pi) + log(rss/n) + 1)

scores_nodo <- function(rss, n, k) {
  ll <- loglik_gauss(rss, n)
  c(aic = ll - k, bic = ll - (log(n)/2)*k)
}

scores_gbn_nodo <- function(nodo, pa, d) {
  n <- nrow(d)
  if (length(pa) == 0) return(scores_nodo(sum((d[[nodo]] - mean(d[[nodo]]))^2), n, 2))
  m <- lm(as.formula(paste(nodo, "~", paste(pa, collapse = "+"))), data = d)
  scores_nodo(sum(residuals(m)^2), n, length(coef(m)) + 1)
}

scores_np_nodo <- function(nodo, pa, d) {
  n <- nrow(d)
  # Las raíces se modelan igual en ambos brazos: entran para que los totales
  # queden en la misma escala que score()
  if (length(pa) == 0) return(scores_nodo(sum((d[[nodo]] - mean(d[[nodo]]))^2), n, 2))
  bw <- sapply(pa, function(p) silverman_bw(d[[p]], d = length(pa)))
  f <- nw_fit(d[, pa, drop = FALSE], d[[nodo]], bw)
  scores_nodo(sum((d[[nodo]] - f$fitted)^2), n, sum(f$hii) + 1)
}
